## 练习 1：Main 页新增 oneof 选项


### 新建 4 个文件（OemJYTianPkg\JYTianSetupModule\）

#### JYTianSetup.sd —— 表单定义

```c
#ifdef SETUP_DATA_DEFINITION
    // ① 数据字段：并入全局 SETUP_DATA 结构体
    //    varstore 锚点：控件 varid = SETUP_DATA.xxx 指向这里的字段
    UINT8 JYTianMyOneOf;
#endif

#ifdef MAIN_FORM_SET

#ifdef OEM_FORM_SET_ITEM
    // ② 控件：Main页第一行直接内嵌（OEM_FORM_SET_ITEM 是 Main 页的 OEM 挂载段）
    //    questionid 用 AUTO_ID 宏，构建时由工具自动分配数值（Build\AutoId.h），无需手工注册
    oneof varid = SETUP_DATA.JYTianMyOneOf,
        questionid = AUTO_ID(JYTIAN_MY_ONEOF_ID),
        prompt     = STRING_TOKEN(STR_JYTIAN_MY_ONEOF_PROMPT),
        help       = STRING_TOKEN(STR_JYTIAN_MY_ONEOF_HELP),
        // option 文本复用全局字符串；默认项 flags 组合是 AMI 惯例（非 edk2 的 flags=DEFAULT）
        option text = STRING_TOKEN(STR_DISABLED), value = 0, flags = DEFAULT | MANUFACTURING | RESET_REQUIRED;
        option text = STRING_TOKEN(STR_ENABLED),  value = 1, flags = RESET_REQUIRED;
    endoneof;
#endif

#endif
```

### 关键机制

| 机制 | 说明 |
| --- | --- |
| SETUP_DATA_DEFINITION | 各模块 SD 的字段段合并进全局 `SETUP_DATA` 结构体 |
| 顶级菜单段共同贡献 | `MAIN_FORM_SET` 可由多个 SD 文件共同贡献（本模块提供 ITEM，main.sd 提供其他内容） |
| AUTO_ID | `AUTO_ID(XXX_ID)` 由构建工具扫描 SD 自动生成数值到 `Build\AutoId.h`，注释带 origin 文件与行号 |
| ELINK | `.sdl` 里 `Parent = SETUP_DEFINITIONS / SetupStringFiles` 把 sd/uni 挂进 Setup 编译 |

## 练习 3：suppressif 条件控制

练习 1 的 oneof 总是显示；练习 3 让第二个选项仅在第一个为 `Enabled` 时出现。

#### JYTianSetup.sd 新增片段

```c
#ifdef SETUP_DATA_DEFINITION
    UINT8 JYTianMyOneOf;
    UINT8 JYTianSubOption;     // 新增字段
#endif

#ifdef OEM_FORM_SET_ITEM
    oneof varid = SETUP_DATA.JYTianMyOneOf, ... endoneof;

    // suppressif 条件控制：仅当 JYTianMyOneOf == 1(Enabled) 时显示
    // 区别于 grayoutif（灰化可见不可操作）
    suppressif ideqval SETUP_DATA.JYTianMyOneOf == 0;
        oneof varid = SETUP_DATA.JYTianSubOption,
            questionid = AUTO_ID(JYTIAN_SUB_OPTION_ID),
            prompt     = STRING_TOKEN(STR_JYTIAN_SUB_OPTION_PROMPT),
            help       = STRING_TOKEN(STR_JYTIAN_SUB_OPTION_HELP),
            option text = STRING_TOKEN(STR_DISABLED), value = 0, flags = DEFAULT | MANUFACTURING | RESET_REQUIRED;
            option text = STRING_TOKEN(STR_ENABLED),  value = 1, flags = RESET_REQUIRED;
        endoneof;
    endif;                      // suppressif 配 endif;（不是 endsuppressif）
#endif
```

#### 验证

Main 页 `JYTian My Oneof` = Enabled 时，"JYTian Sub Option" 出现；= Disabled 时整行隐藏（不在屏幕渲染）。

## 练习 4：新 Form + goto 链接

在 Main 页底部加跳转入口，进入独立的子配置页。

#### JYTianSetup.sd 新增片段

```c
#ifdef OEM_FORM_SET_GOTO
    // Main 页的 goto 段名是 OEM_FORM_SET_GOTO（不是 FORM_SET_GOTO）
    // goto 目标用 formid 直接写宏名：AUTO_ID(x) 在 Build\AutoId.h 里就是 x 本身
    goto JYTIAN_SUB_FORM_ID,
        prompt = STRING_TOKEN(STR_JYTIAN_SUB_FORM_PROMPT),
        help   = STRING_TOKEN(STR_JYTIAN_SUB_FORM_HELP);
#endif

#ifdef FORM_SET_FORM
    form formid = AUTO_ID(JYTIAN_SUB_FORM_ID),
        title = STRING_TOKEN(STR_JYTIAN_SUB_FORM_TITLE);
        SUBTITLE(STRING_TOKEN(STR_JYTIAN_SUB_FORM_TITLE))
        SEPARATOR

        checkbox varid = SETUP_DATA.JYTianSubFormFlag,
            questionid = AUTO_ID(JYTIAN_SUB_FORM_FLAG_ID),
            prompt = STRING_TOKEN(STR_JYTIAN_SUB_FORM_FLAG_PROMPT),
            help   = STRING_TOKEN(STR_JYTIAN_SUB_FORM_FLAG_HELP),
            default = 0,    // checkbox 属性行必须用逗号；分号会给 option
        endcheckbox;        // checkbox 必须显式 endcheckbox; 闭合

        SEPARATOR
        // 练习 5 的 efivarstore 控件放这里
    endform;
#endif
```

## 练习 5：efivarstore 独立存储

练习 1 字段存在全局 `SETUP_DATA`；练习 5 用独立 GUID 的 EFI 变量，重启后值保持。

#### JYTianVarStore.h（新建）

```c
#ifndef _JYTIAN_VAR_STORE_H__
#define _JYTIAN_VAR_STORE_H__

// 独立 EFI 变量的 GUID；与 SETUP_DATA 完全无关
#define JYTIAN_EFI_VARSTORE_GUID \
{ 0x2bfd086d, 0x16b9, 0x47fb, { 0xa6, 0x01, 0x70, 0xf0, 0x59, 0xce, 0x02, 0xff } }

#pragma pack(1)
typedef struct _JYTIAN_EFI_VARSTORE {
  UINT8 JYTianEfiValue;
} JYTIAN_EFI_VARSTORE;
#pragma pack()
#endif
```

#### JYTianSetup.sd 新增片段

```c
#ifdef FORM_SET_TYPEDEF
    #include "JYTianVarStore.h"      // 引入 GUID 与结构体定义
#endif

#ifdef FORM_SET_VARSTORE
    // efivarstore 是 edk2 关键字：attribute + name + guid
    // attribute = 0x07 = NV | BS | RT（OS 下也可读；只要 BS+NV 用 0x03）
    efivarstore JYTIAN_EFI_VARSTORE,
        attribute = 0x07,
        name  = JYTianEfiVar,
        guid  = JYTIAN_EFI_VARSTORE_GUID;
#endif
```

子页内的控件（`varid = JYTianEfiVar.JYTianEfiValue` 是 varstore 名.字段名）：

```c
        oneof varid = JYTianEfiVar.JYTianEfiValue,
            questionid = AUTO_ID(JYTIAN_EFI_VALUE_ID),
            prompt     = STRING_TOKEN(STR_JYTIAN_EFI_VALUE_PROMPT),
            help       = STRING_TOKEN(STR_JYTIAN_EFI_VALUE_HELP),
            flags      = INTERACTIVE,    // 练习 6：触发回调
            option text = STRING_TOKEN(STR_DISABLED), value = 0, flags = DEFAULT | MANUFACTURING | RESET_REQUIRED;
            option text = STRING_TOKEN(STR_ENABLED),  value = 1, flags = RESET_REQUIRED;
        endoneof;
```

## 练习 6：Callback 处理 QuestionId

修改 `JYTianEfiValue` 时触发回调，把它同步到 Main 页的 `JYTianSubFormFlag`（验证联动效果）。

#### JYTianSetupCallBack.c（新建）

```c
#include <Efi.h>
#include <AmiLib.h>
#include <AmiDxeLib.h>
#include <Setup.h>
#include <Token.h>
#include <Library\HiiLib.h>

EFI_STATUS
JYTianSetupCallback (
    IN EFI_HII_HANDLE HiiHandle,
    IN UINT16         Class,
    IN UINT16         SubClass,
    IN UINT16         Key )
{
    SETUP_DATA          *SetupData = NULL;
    EFI_STATUS          Status;
    EFI_GUID            SetupGuid = SETUP_GUID;
    UINTN               BufferSize = sizeof (SETUP_DATA);
    CALLBACK_PARAMETERS *CallbackParameter;

    // 只处理值改变事件，其他事件不响应
    CallbackParameter = GetCallbackParameters ();
    if (CallbackParameter->Action != EFI_BROWSER_ACTION_CHANGING) {
        return EFI_UNSUPPORTED;
    }

    switch (Key) {
    case JYTIAN_EFI_VALUE_ID:
        Status = pBS->AllocatePool (EfiBootServicesData, BufferSize, (VOID **)&SetupData);
        if (EFI_ERROR (Status)) return Status;
        Status = HiiLibGetBrowserData (&BufferSize, SetupData, &SetupGuid, L"Setup");
        if (EFI_ERROR (Status)) { pBS->FreePool (SetupData); return Status; }

        // 联动：把 Efi Value 新值同步到 Main 页的 Sub Form Flag
        SetupData->JYTianSubFormFlag = (UINT8)(CallbackParameter->Value->u8 == 1);

        Status = HiiLibSetBrowserData (BufferSize, SetupData, &SetupGuid, L"Setup");
        pBS->FreePool (SetupData);
        return Status;
    default:
        return EFI_UNSUPPORTED;
    }
}
```

#### JYTianSetupModule.sdl 新增 ELINK

```
ELINK
    Name  = "$(JYTIAN_SETUP_DIR)/JYTianSetupCallBack.c"
    Parent  = "SetupCallbackFiles"        // 把 .c 挂进 SetupDxe 编译
    InvokeOrder = AfterParent
End

// ITEM_CALLBACK 注册：CLASS, SubClass, QId, Fn
// CLASS 用 MAIN_FORM_SET_CLASS（Setup.h:40）—— Main 页所属 CLASS
ELINK
    Name  = "ITEM_CALLBACK(MAIN_FORM_SET_CLASS,0,JYTIAN_EFI_VALUE_ID,JYTianSetupCallback),"
    Parent  = "SetupItemCallbacks"
    InvokeOrder = AfterParent
End
```

#### 验证

子页 `JYTian Efi Value` 从 Disabled 切到 Enabled → 回 Main 页，`JYTian Sub Form Flag` 已同步为 Enabled（联动生效）。

## 排错记录（VfrCompile SD 语法）

| 现象 | 根因 | 修复 |
| --- | --- | --- |
| 报 `subtitle/questionid/endoneof : unexpected token` | checkbox 块缺 `endcheckbox;` 闭合 | 补 `endcheckbox;` |
| 补后仍报 `endcheckbox : unexpected token` | checkbox 的 `default` 属性用了分号 `default = 0;` | 改成逗号 `default = 0,` |

**SD/VFR 块语句结尾规则**：

| 元素 | 属性行 | option 行 | 块结束 |
| --- | --- | --- | --- |
| oneof | 逗号 | 分号 | `endoneof;` |
| checkbox | 逗号 | — | `endcheckbox;` |
| form | 标题分号 | — | `endform;` |
| suppressif | 条件分号 | — | `endif;` |
