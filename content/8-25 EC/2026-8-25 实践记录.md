写一个UEFI Application 在shell下读出EC 版本

EC 对主机暴露两个 I/O 端口：

| 端口 | 方向 | 作用 |
|---|---|---|
| 0x66 | 读 | 读状态寄存器 |
| 0x66 | 写 | 写命令字节（Read RAM / Query 等） |
| 0x62 | 读 | 读数据（EC 返回的字节） |
| 0x62 | 写 | 写数据字节（RAM 偏移地址、参数） |


主要实现思路：

### 1. WaitEcObf —— 等 EC 返回数据，防止数据覆盖丢失

```c
#define EC_CMD_PORT     0x66   // 命令/状态端口
#define EC_DATA_PORT    0x62   // 数据端口

EFI_STATUS
WaitEcObf (
  UINT16  TimeOut              // 超时计数上限，传入 1000
  )
{
  UINT16  Count = 0;           // 循环计数器
  UINT8   RegStat;

  RegStat = IoRead8 (EC_CMD_PORT);      // 读状态寄存器
  while ((RegStat & 0x01) == 0) {       // bit0(OBF)=0 说明 EC 还没返回数据
    RegStat = IoRead8 (EC_CMD_PORT);         // 反复重读状态
    Count++;                                 // 计一次轮询
    if (Count == TimeOut) {             // 等满上限仍没 OBF → 设备错误
      return EFI_DEVICE_ERROR;
    }
  }
  return EFI_SUCCESS;                        // 等到 OBF=1，可读数据
}
```

纯忙等，无延时无中断。`TimeOut` 是循环次数不是毫秒，值由调用方拍脑袋定（这里传 1000）。EC 事务通常几十微秒完成，忙等可接受。

### 2. WaitEcIbe —— 等输入缓冲可写

写命令/数据的**前置条件**。

```c
EFI_STATUS
WaitEcIbe (
  UINT16  TimeOut
  )
{
  UINT16  Count = 0;
  UINT8   RegStat;

  MicroSecondDelay (200);                    // 进循环前先延时 200us 给 EC 反应
  RegStat = IoRead8 (EC_CMD_PORT);           // 读状态寄存器
  while ((RegStat & 0x02) != 0) {            // bit1 不为 0 时继续等（缓冲仍占用）
    RegStat = IoRead8 (EC_CMD_PORT);
    Count++;
    if (Count == TimeOut) {
      return EFI_DEVICE_ERROR;
    }
  }
  return EFI_SUCCESS;
}
```

`MicroSecondDelay (200)` 只在 while 之前执行一次，循环体内无延时。先给 200us 缓冲时间，随后转入忙等。

### 3. EcRead —— 单字节读完整事务

```c
#define EC_CMD_READ_RAM  0x80          // ACPI EC 标准命令：Read EC RAM

UINT8
EcRead (
  UINT8  Offset                // EC RAM 偏移地址
  )
{
  UINT8   Buffer = 0xFF;       // 默认值，读失败时保持 0xFF 供调用方识别
  UINT32  TimeOut = 0;

  // 第 1 步：排空输出缓冲。若有之前的数据残留，就先读走，防止数据污染
  while ((IoRead8 (EC_CMD_PORT) & 0x01) != 0x00) {
    IoRead8 (EC_DATA_PORT);    // 读数据端口，丢弃
    TimeOut++;
    if (TimeOut == 1000) {     // 最多清 1000 次，防死循环
      break;
    }
  }

  // 第 2 步：写命令字节 0x80 (Read RAM) 到命令端口
  if (WaitEcIbe (1000) == EFI_SUCCESS) {     // 先等输入缓冲可写
    MicroSecondDelay (200);                  // 再给一点缓冲
    IoWrite8 (EC_CMD_PORT, EC_CMD_READ_RAM); // 发命令
  }

  // 第 3 步：写偏移地址到数据端口
  if (WaitEcIbe (1000) == EFI_SUCCESS) {
    MicroSecondDelay (200);
    IoWrite8 (EC_DATA_PORT, Offset);         // 告诉 EC 从哪个偏移取
    WaitEcIbe (1000);                        // 给 EC 时间处理地址（这里没检查返回值）
  }

  // 第 4 步：等 EC 把数据放进输出缓冲，再读回
  if (WaitEcObf (1000) == EFI_SUCCESS) {
    Buffer = IoRead8 (EC_DATA_PORT);         // 真正读到版本字节
  }

  return Buffer;                             // 失败时返回默认 0xFF
}
```

先排空输出缓冲是 EC 端口访问最常见的隐性 bug：前一次事务若没把 OBF 数据读干净（bit0 还挂着），下次一进 `WaitEcObf` 会立即返回成功，读到的却是旧字节。先清空，保证 `WaitEcObf` 等到的是本次事务的结果。

### 4. ReadEcAppMain —— 主流程

UEFI_APPLICATION 标准入口，`EFIAPI` + `gBS` 由工具链约定。

```c
#define EC_VERSION_MAIN   0x00   // EC RAM 偏移 0x00：主版本
#define EC_VERSION_SUB    0x01   // EC RAM 偏移 0x01：次版本
#define EC_VERSION_TEST   0x02   // EC RAM 偏移 0x02：测试版本

EFI_STATUS
EFIAPI
ReadEcAppMain (
  IN EFI_HANDLE        ImageHandle,      // 本镜像句柄
  IN EFI_SYSTEM_TABLE  *SystemTable      // 系统表指针
  )
{
  UINT8  MainVersion;   // 主版本字节
  UINT8  SubVersion;    // 次版本字节
  UINT8  TestVersion;   // 测试版本字节

  Print (L"[ReadEcApp] Reading EC firmware version...\n");

  // 三次单字节读，取 EC RAM 最前面三个字节
  MainVersion = EcRead (EC_VERSION_MAIN);
  SubVersion  = EcRead (EC_VERSION_SUB);
  TestVersion = EcRead (EC_VERSION_TEST);

  // 十进制 + 十六进制同时打印，方便肉眼核对
  Print (
    L"[ReadEcApp] EC FW version = %d.%d.%d  (0x%02X.0x%02X.0x%02X)\n",
    MainVersion,
    SubVersion,
    TestVersion,
    MainVersion,
    SubVersion,
    TestVersion
  );

  return EFI_SUCCESS;   // 返回成功，APP 退出后回到 Shell
}
```
