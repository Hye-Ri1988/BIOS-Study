# UsbPower UEFI APP 实践记录
 目标：写一个 UEFI APP，在 UEFI Shell 下控制 USB 电源开关（能开能关）

## 一、硬件

- 控制引脚：**GPP_D16**（pad 编号 `0x08110010`），网络名 **PWREN_USBBP1**（背板 USB1 电源使能）
- **高电平使能**：GPIO 输出高 = USB 通电，输出低 = 断电（实测确认）
- 链路：GPP_D16 → JW7115S 限流开关 EN → USB 端口 VBUS

## 二、实现思路

```
UsbPower.efi ON/OFF/STATUS
    ├─ 入口 UsbPowerMain（EFI_APPLICATION_ENTRY_POINT）
    ├─ 取命令行参数：EFI_LOADED_IMAGE_PROTOCOL->LoadOptions
    ├─ 解析 on/off/status
    └─ GpioLib API 控制：
         ON    → GpioSetOutputValue(GPP_D16, 1)
         OFF   → GpioSetOutputValue(GPP_D16, 0)
         STATUS→ GpioGetOutputValue 读回电平
```


## 三、关键代码

### 3.1 入口与参数解析

Shell 传参在 `EFI_LOADED_IMAGE_PROTOCOL->LoadOptions`（CHAR16 字符串）。**注意 Shell 传的格式不统一**：有的含文件名（`UsbPower.efi OFF`），有的只有参数（`OFF`）。所以不能用"跳过第一个 token"的写法，直接对整个字符串做关键词匹配最稳。

```c
// 解析命令行，返回 0=状态 1=开 2=关
STATIC
UINTN
UsbPowerParseArg (
  IN EFI_HANDLE  ImageHandle
  )
{
  EFI_LOADED_IMAGE_PROTOCOL  *LoadedImage;
  CHAR16                     *LoadOptions;
  CHAR16                     *Lower;
  CHAR16                     *p;
  UINTN                      Length;

  // 通过 LoadedImage Protocol 拿命令行参数
  gBS->OpenProtocol (
         ImageHandle,
         &gEfiLoadedImageProtocolGuid,
         (VOID **)&LoadedImage,
         ImageHandle,
         NULL,
         EFI_OPEN_PROTOCOL_GET_PROTOCOL
         );

  if (LoadedImage == NULL || LoadedImage->LoadOptions == NULL) {
    return 0;  // 无参数 = 状态
  }
  LoadOptions = (CHAR16 *)LoadedImage->LoadOptions;
  Print (L"LoadOptions: \"%s\"\n", LoadOptions);  // 调试：看 Shell 实际传了什么

  // 复制到局部缓冲并转小写
  Length = StrLen (LoadOptions);
  if (Length > 63) Length = 63;
  Lower = AllocatePool ((Length + 1) * sizeof (CHAR16));
  if (Lower == NULL) return 0;
  for (p = LoadOptions; p < LoadOptions + Length; p++) {
    Lower[p - LoadOptions] = (*p >= L'A' && *p <= L'Z') ? (*p + 32) : *p;
  }
  Lower[Length] = L'\0';

  // 关键词匹配，OFF 在前（off 不含 on，无冲突）
  if (StrStr (Lower, L"off") != NULL) {
    FreePool (Lower);
    return 2;
  }
  if (StrStr (Lower, L"on") != NULL) {
    FreePool (Lower);
    return 1;
  }
  FreePool (Lower);
  return 0;
}
```

### 3.2 GPIO 控制主逻辑

```c
EFI_STATUS
EFIAPI
UsbPowerMain (
  IN EFI_HANDLE        ImageHandle,
  IN EFI_SYSTEM_TABLE  *SystemTable
  )
{
  UINTN   Action;
  EFI_STATUS  Status;

  Action = UsbPowerParseArg (ImageHandle);

  if (Action == 1) {
    // 开：输出高（高电平使能）
    Status = GpioSetOutputValue (USB_PWR_GPIO, 1);
    if (EFI_ERROR (Status)) {
      Print (L"Power on failed Status=%r\n", Status);
      return EFI_DEVICE_ERROR;
    }
    Print (L"USB power is ON\n");
  } else if (Action == 2) {
    // 关：输出低
    Status = GpioSetOutputValue (USB_PWR_GPIO, 0);
    if (EFI_ERROR (Status)) {
      Print (L"Power off failed Status=%r\n", Status);
      return EFI_DEVICE_ERROR;
    }
    Print (L"USB power is OFF\n");
  } else {
    // 无参数 / STATUS：读回电平显示
    UsbPowerShowStatus ();
    UsbPowerShowHelp ();
  }

  return EFI_SUCCESS;
}
```

读状态用 `GpioGetOutputValue`（读输出电平），失败时回退读 `GpioGetPadConfig` 的 `OutputState`：

```c
STATIC
VOID
UsbPowerShowStatus (
  VOID
  )
{
  EFI_STATUS   Status;
  GPIO_CONFIG  GpioCfg;
  UINT32       Value;

  Value = 0;
  Status = GpioGetOutputValue (USB_PWR_GPIO, &Value);
  if (EFI_ERROR (Status)) {
    // 回退：读输出配置里的 OutputState
    GpioGetPadConfig (USB_PWR_GPIO, &GpioCfg);
    Value = (GpioCfg.OutputState == GpioOutHigh) ? 1 : 0;
  }

  Print (L"USB power (GPP_D16): %s\n", Value ? L"ON" : L"OFF");
}
```

### 3.3 INF 关键段

```inf
[Defines]
  INF_VERSION     = 0x00010005
  BASE_NAME       = UsbPower
  MODULE_TYPE     = UEFI_APPLICATION
  FILE_GUID       = 1DFB6DAE-0D00-49E9-A8F7-2029873E681E
  ENTRY_POINT     = UsbPowerMain

[Packages]
  MdePkg/MdePkg.dec
  OemJYTianPkg/OemJYTianPkg.dec
  ClientOneSiliconPkg/SiPkg.dec      ; GpioLib 头文件所在包

[LibraryClasses]
  UefiApplicationEntryPoint
  UefiBootServicesTableLib
  UefiLib
  PrintLib
  GpioLib

[Protocols]
  gEfiLoadedImageProtocolGuid        ; 协议 GUID 必须放 [Protocols]，不是 [Guids]
```
